# 0-10 setup: two PostgreSQL instances. A = be-postgres (make up, host 5432); B = r010-pg-b (host 25432).
# demo/east uses A, demo/west uses B; both really connect (pgx) and report which instance they reached (r010_meta.instance).

EAST_PW='east-pass-1'
WEST_PW='We$t"pa\ss{1}'

# instance B
docker rm -f r010-pg-b >/dev/null 2>&1 || true
docker run -d --name r010-pg-b -e POSTGRES_PASSWORD=r010-admin -p 25432:5432 postgres:16-alpine >/dev/null
for _ in $(seq 1 60); do docker exec r010-pg-b pg_isready -U postgres >/dev/null 2>&1 && break; sleep 0.5; done; sleep 1
prep() { # <container> <instance-name> <role> <password>
  docker exec -i "$1" psql -q -U postgres -v ON_ERROR_STOP=1 <<SQL
DROP DATABASE IF EXISTS r010;
DROP ROLE IF EXISTS $3;
CREATE ROLE $3 LOGIN PASSWORD \$pw\$$4\$pw\$;
CREATE DATABASE r010;
\c r010
CREATE SCHEMA r010_meta;
CREATE TABLE r010_meta.instance(name text);
INSERT INTO r010_meta.instance VALUES ('$2');
GRANT USAGE ON SCHEMA r010_meta TO $3;
GRANT SELECT ON r010_meta.instance TO $3;
GRANT CREATE ON DATABASE r010 TO $3;
SQL
}

// Package $n is a stub member that talks to its own PostgreSQL database, configured only from its config table.
package $n

import (
	"context"
	"encoding/json"
	"net"
	"net/http"
	"net/url"
	"sync"

	"github.com/jackc/pgx/v5/pgxpool"
)

// Version is the version of this module's code.
const Version = "0.1.0"

// ConnString builds the connection string from the config table; the password may hold any character.
func ConnString(cfg map[string]string) string {
	u := url.URL{Scheme: "postgres", User: url.UserPassword(cfg["PG_USER"], cfg["PG_PASSWORD"]),
		Host: net.JoinHostPort(cfg["PG_HOST"], cfg["PG_PORT"]), Path: "/" + cfg["PG_DATABASE"]}
	return u.String()
}

// Migrate creates this component's schema and table, as its own role.
func Migrate(ctx context.Context, cfg map[string]string) (string, error) {
	pool, err := pgxpool.New(ctx, ConnString(cfg))
	if err != nil {
		return "", err
	}
	defer pool.Close()
	var inst string
	err = pool.QueryRow(ctx, \`CREATE SCHEMA IF NOT EXISTS $n;
CREATE TABLE IF NOT EXISTS $n.hello(id int PRIMARY KEY, at timestamptz DEFAULT now());
INSERT INTO $n.hello(id) VALUES (1) ON CONFLICT DO NOTHING;
SELECT name FROM r010_meta.instance\`).Scan(&inst)
	if err != nil {
		// pgx runs a multi-statement string through the simple protocol only without arguments; fall back to one by one
		for _, q := range []string{"CREATE SCHEMA IF NOT EXISTS $n",
			"CREATE TABLE IF NOT EXISTS $n.hello(id int PRIMARY KEY, at timestamptz DEFAULT now())",
			"INSERT INTO $n.hello(id) VALUES (1) ON CONFLICT DO NOTHING"} {
			if _, err := pool.Exec(ctx, q); err != nil {
				return "", err
			}
		}
		if err := pool.QueryRow(ctx, "SELECT name FROM r010_meta.instance").Scan(&inst); err != nil {
			return "", err
		}
	}
	return inst, nil
}

// Handler builds the member's HTTP handler from its config table, never from the process environment.
func Handler(cfg map[string]string) http.Handler {
	var once sync.Once
	var pool *pgxpool.Pool
	var poolErr error
	mux := http.NewServeMux()
	mux.HandleFunc("/healthz", func(w http.ResponseWriter, _ *http.Request) { w.Write([]byte("ok")) })
	mux.HandleFunc("/api/db", func(w http.ResponseWriter, r *http.Request) {
		once.Do(func() { pool, poolErr = pgxpool.New(context.Background(), ConnString(cfg)) })
		if poolErr != nil {
			http.Error(w, poolErr.Error(), http.StatusBadGateway)
			return
		}
		var db, user, inst, rows string
		err := pool.QueryRow(r.Context(), \`SELECT current_database(), current_user, (SELECT name FROM r010_meta.instance),
			(SELECT count(*)::text FROM $n.hello)\`).Scan(&db, &user, &inst, &rows)
		if err != nil {
			http.Error(w, err.Error(), http.StatusBadGateway)
			return
		}
		_ = json.NewEncoder(w).Encode(map[string]string{"component": "demo/$n", "instance": inst, "database": db,
			"user": user, "rowsInOwnSchema": rows, "addr": net.JoinHostPort(cfg["PG_HOST"], cfg["PG_PORT"])})
	})
	return mux
}
EOF

// Command dbshell hosts the members it compiles in, as told by BRICKKIT_SERVED_MEMBERS_CONFIG.
package main

import (
	"encoding/json"
	"fmt"
	"log"
	"net/http"
	"os"

	"example.test/east"
	"example.test/west"
)

type member struct {
	ComponentID string            `json:"componentId"`
	Version     string            `json:"version"`
	HTTPPort    int               `json:"httpPort"`
	Config      map[string]string `json:"config"`
}

var compiled = map[string]func(map[string]string) http.Handler{"demo/east": east.Handler, "demo/west": west.Handler}

func main() {
	var members []member
	if err := json.Unmarshal([]byte(os.Getenv("BRICKKIT_SERVED_MEMBERS_CONFIG")), &members); err != nil {
		log.Fatalf("BRICKKIT_SERVED_MEMBERS_CONFIG: %v", err)
	}
	for _, m := range members {
		h, ok := compiled[m.ComponentID]
		if !ok {
			log.Fatalf("this shell has no module %s", m.ComponentID)
		}
		addr := fmt.Sprintf(":%d", m.HTTPPort)
		go func(h http.Handler) { log.Fatal(http.ListenAndServe(addr, h)) }(h(m.Config))
		log.Printf("serving %s@%s on %s", m.ComponentID, m.Version, addr)
	}
	port := os.Getenv("PORT")
	if port == "" {
		port = "8100"
	}
	http.HandleFunc("/healthz", func(w http.ResponseWriter, _ *http.Request) { w.Write([]byte("ok")) })
	log.Fatal(http.ListenAndServe(":"+port, nil))
}
EOF
