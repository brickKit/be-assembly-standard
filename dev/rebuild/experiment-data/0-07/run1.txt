// infra dials PostgreSQL and NATS at the addresses in the config table.
func infra(cfg map[string]string) map[string]string {
	out := map[string]string{}
	dial := func(name, addr string) {
		c, err := net.DialTimeout("tcp", addr, 2*time.Second)
		if err != nil {
			out[name] = addr + " FAIL " + err.Error()
			return
		}
		c.Close()
		out[name] = addr + " ok"
	}
	dial("pg", net.JoinHostPort(cfg["PG_HOST"], cfg["PG_PORT"]))
	if u, err := url.Parse(cfg["NATS_URL"]); err == nil {
		dial("nats", u.Host)
	}
	return out
}'

import json

$PY_COMMON

class Handler(Base):
    def do_GET(self):
        if self.path == "/healthz":
            return self.send_json(200, "ok")
        if self.path == "/api/infra":
            return self.send_json(200, infra(cfg))
        if self.path == "/api/pid":
            return self.send_json(200, {"pid": os.getpid()})
        if self.path == "/api/chain":
            a, p = cfg.get("DEMO_ALPHA_ENDPOINT", ""), cfg.get("DEMO_PYR_ENDPOINT", "")
            try:

            except OSError as e:
                return self.send_json(502, {"error": str(e)})
        self.send_json(404, {"error": self.path})
