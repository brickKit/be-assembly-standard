"""Python shell: hosts the members named in BRICKKIT_SERVED_MEMBERS_CONFIG."""
import json
import os
import sys
import threading
from http.server import BaseHTTPRequestHandler, ThreadingHTTPServer

# From source the member's own directory; in the image its vendored copy (as the Go shell vendors its members).
HERE = os.path.dirname(os.path.abspath(__file__))
SOURCE = os.path.join(HERE, "../../../components/demo/pyr")
sys.path.insert(0, SOURCE if os.path.isdir(SOURCE) else os.path.join(HERE, "vendor"))
import pyr  # noqa: E402

COMPILED = {"demo/pyr": pyr}

raw = os.environ.get("BRICKKIT_SERVED_MEMBERS_CONFIG")
if raw is None:
    sys.exit("BRICKKIT_SERVED_MEMBERS_CONFIG is not set")
for m in json.loads(raw):
    mod = COMPILED.get(m["componentId"])
    if mod is None:
        sys.exit(f"this shell has no module {m['componentId']}")
    if mod.VERSION != m["version"]:
        sys.exit(f"asked to host {m['componentId']}@{m['version']}, but the code is {mod.VERSION}")
    srv = ThreadingHTTPServer(("0.0.0.0", m["httpPort"]), mod.handler(m.get("config", {})))
    threading.Thread(target=srv.serve_forever, daemon=True).start()
    print(f"serving {m['componentId']}@{m['version']} on :{m['httpPort']}", flush=True)

        pass

    def do_GET(self):
        self.send_response(200 if self.path == "/healthz" else 404)
        self.end_headers()
