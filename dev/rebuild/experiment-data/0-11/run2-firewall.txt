######## run011-fw.sh
#!/usr/bin/env bash
# 0-11 follow-up (run with sudo): why rootful Podman containers cannot reach each other on this host.
# 1. read the firewall (FORWARD policy and rules, iptables and nft) and Podman's network backend;
# 2. two throwaway containers on a rootful Podman network: call one from the other, watch the FORWARD drop counters;
# 3. temporarily accept forwarded traffic from Podman's bridge in DOCKER-USER, call again, then delete that rule.
# Everything it adds is removed at the end. Log: /tmp/r011-fw.log (owned by the invoking user).
set -uo pipefail
[ "$(id -u)" = 0 ] || { echo "run me with sudo"; exit 1; }
LOG=/tmp/r011-fw.log; OWNER=${SUDO_USER:-zhijie}
exec > >(tee "$LOG") 2>&1
sec() { echo; echo "######## $*"; }
run() { echo "\$ $*"; "$@" 2>&1; echo "[exit $?]"; }
drops() { iptables -L FORWARD -v -n -x 2>/dev/null | awk 'NR==1{print "  policy line: " $0}'; }
cleanup() { podman rm -f r011-fw-a r011-fw-b >/dev/null 2>&1; podman network rm -f r011-fw >/dev/null 2>&1
  iptables -D DOCKER-USER -i r011fw0 -j ACCEPT 2>/dev/null; iptables -D DOCKER-USER -o r011fw0 -j ACCEPT 2>/dev/null; }
trap cleanup EXIT
echo "# 0-11 firewall check ($(date '+%F %T')), $(podman --version), $(docker --version 2>/dev/null | cut -d, -f1), $(iptables --version)"

sec "1. firewall and Podman network backend (read only)"
run sysctl net.bridge.bridge-nf-call-iptables net.ipv4.ip_forward
run iptables -S FORWARD
run iptables -L FORWARD -v -n --line-numbers
run iptables -S DOCKER-USER
run podman info --format 'networkBackend={{.Host.NetworkBackend}}'
run podman info --format '{{json .Host.NetworkBackendInfo}}'
echo "\$ nft list ruleset | chains with a drop policy, and netavark's tables"
nft list ruleset 2>&1 | grep -n -E "^table|chain .*\{|policy drop|netavark" | head -60; echo "[exit $?]"

sec "2. two throwaway containers on a rootful network (bridge r011fw0): b calls a"
cleanup
run podman network create --interface-name r011fw0 r011-fw
run podman run -d --name r011-fw-a --network r011-fw docker.io/library/python:3.13-alpine python3 -m http.server 8080
sleep 3
echo "FORWARD counters before:"; iptables -L FORWARD -v -n -x | head -3
run podman run --rm --name r011-fw-b --network r011-fw docker.io/library/alpine:3 wget -qO- -T 4 http://r011-fw-a:8080/
echo "FORWARD counters after:"; iptables -L FORWARD -v -n -x | head -3
echo "\$ podman exec r011-fw-a wget -qO- -T 2 http://127.0.0.1:8080/ (inside the same container: no forwarding involved)"
podman exec r011-fw-a wget -qO- -T 2 http://127.0.0.1:8080/ | head -2; echo "[exit $?]"

sec "3. temporarily accept forwarded traffic of r011fw0 in DOCKER-USER, call again, then remove the rules"
run iptables -I DOCKER-USER -i r011fw0 -j ACCEPT
run iptables -I DOCKER-USER -o r011fw0 -j ACCEPT
run podman run --rm --name r011-fw-b --network r011-fw docker.io/library/alpine:3 wget -qO- -T 4 http://r011-fw-a:8080/
run iptables -D DOCKER-USER -i r011fw0 -j ACCEPT
run iptables -D DOCKER-USER -o r011fw0 -j ACCEPT
run iptables -S DOCKER-USER

sec "cleanup"
cleanup; trap - EXIT
echo "containers and network removed; DOCKER-USER back to:"; iptables -S DOCKER-USER
chown "$OWNER" "$LOG"; echo "log: $LOG"
