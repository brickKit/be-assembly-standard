#!/usr/bin/env bash
# 0-09 run 1: two replicas of one component — Kubernetes (minikube), and what Docker does with replicas. Uses the 0-07 project.
set -uo pipefail
S=$(dirname "$0")/..; S=$(cd "$S" && pwd); P=$S/r007/proj; export GOCACHE=$S/gocache; export PATH=$S/k8sbin:$PATH; NS=r007
run() { echo "\$ $*"; "$@" 2>&1 | sed -E 's/"time":"[^"]*"/"time":"…"/'; echo "[exit ${PIPESTATUS[0]}]"; }
kx() { local d=$1; shift; echo "\$ [in deploy/$d] $*"; kubectl -n $NS exec deploy/$d -- "$@" 2>&1; echo; }
# probe <deploy> <url> <seconds>: one request every 0.2 s from inside a pod; prints how many failed
probe() { kubectl -n $NS exec deploy/$1 -- sh -c "ok=0; bad=0; end=\$((\$(date +%s)+$3)); while [ \$(date +%s) -lt \$end ]; do if wget -qO- -T 2 $2 >/dev/null 2>&1; then ok=\$((ok+1)); else bad=\$((bad+1)); fi; sleep 0.2; done; echo \"ok=\$ok failed=\$bad\""; }
cd "$P"
echo "# 0-09 run 1 ($(date +%F)), $(brickkit version | head -1), $(minikube version | head -1)"
minikube start -p minikube >/dev/null 2>&1; echo "minikube: $(minikube status -f '{{.Host}} {{.APIServer}}')"
kubectl apply -f $S/bin/r007-base.yaml >/dev/null; kubectl -n r007-base rollout status deploy/pg deploy/nats --timeout=180s >/dev/null && echo "base resources ready"
for i in $(docker images --format '{{.Repository}}:{{.Tag}}' --filter reference='demo-*'); do minikube image load $i >/dev/null 2>&1; done; echo "images loaded"

echo; echo "######## part 1 (k8s): delta (standalone) and the Go shell (alpha, beta) with replicas: 2"
cat > deploy.yaml <<'YAML'
target: k8s

k8s:
  context: minikube
  namespace: r007
  createNamespace: true

vars:
  PG_HOST: pg.r007-base.svc.cluster.local
  NATS_URL: nats://nats.r007-base.svc.cluster.local:4222

components:
  - id: demo/delta
    replicas: 2
  - id: demo/gamma
  - id: demo/pyshell
    members:
      - id: demo/pyr
  - id: demo/shell
    replicas: 2
    members:
      - id: demo/alpha
      - id: demo/beta
YAML
cat deploy.yaml
run brickkit lint --all
run brickkit up
run ls .brickkit/generated/k8s/ .brickkit/generated/k8s/*/
for f in .brickkit/generated/k8s/*/*pdb* .brickkit/generated/k8s/*pdb* ; do [ -f "$f" ] && { echo "=== $f"; grep -v "^#" "$f"; }; done 2>/dev/null

echo; echo "######## part 2 (k8s): delete one replica while calls keep coming"
D=$(kubectl -n $NS get pods -l app=demo-delta-0-1-0 -o name | head -1); SH=$(kubectl -n $NS get pods -l app=demo-shell-0-1-0 -o name | head -1)
( sleep 3; echo "deleting $D and $SH"; kubectl -n $NS delete $D $SH --wait=false ) &

wait
sleep 2; run kubectl -n $NS get pods
run brickkit status
