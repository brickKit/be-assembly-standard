#!/usr/bin/env bash
# 0-08 setup: two versions of demo/alpha side by side.
# alpha lives in its own git repository (a bare "remote" remotes/demo-alpha, tags 0.1.0 and 0.2.0), reached through a
# git install source with baseUrl (a component at a repository's root has tags without a namespace, so not `repo`);
# beta depends on alpha@0.1.0, gamma on alpha@0.2.0 (and on beta). The Go shell hosts alpha@0.1.0 and beta.
set -euo pipefail
S=<scratchpad>
bash "$S/bin/r008-base.sh" >/dev/null           # the 0-06 project, regenerated under r008/
# the remote is recreated with new commits each time: drop BrickKit's user-level cache of it (else VERSION_CHANGED, correctly)
rm -rf "$HOME/.cache/brickkit/repos/local$S/r008/remotes"
cd "$S/r008"
git() { command git -c user.name=r008 -c user.email=r008@example.test -c init.defaultBranch=main "$@"; }

# alpha's repository: 0.1.0, then 0.2.0 (version in component.yaml and in the code)
mkdir -p remotes src && git init -q --bare remotes/demo-alpha
cp -r proj/components/demo/alpha src/alpha && cd src/alpha
git init -q && git add -A && git commit -qm "alpha 0.1.0" && git tag 0.1.0
sed -i 's/^  version: 0.1.0/  version: 0.2.0/' component.yaml; sed -i 's/const Version = "0.1.0"/const Version = "0.2.0"/' alpha.go
git commit -qam "alpha 0.2.0" && git tag 0.2.0
git remote add origin "$S/r008/remotes/demo-alpha" && git push -q origin main --tags
cd "$S/r008"; git clone -q --branch 0.1.0 remotes/demo-alpha src/alpha-0.1.0 2>/dev/null

cd proj
rm -rf components/demo/alpha components/demo/delta components/demo/pyr shell/demo/pyshell config
mkdir -p config && printf '# config/vars.yaml — the base resources started by make up\nPG_HOST: host.docker.internal\nPG_PORT: "5432"\nNATS_URL: nats://host.docker.internal:4222\n' > config/vars.yaml
# gamma: depends on alpha@0.2.0 and beta@0.1.0; /api/chain calls both
python3 - <<'PY'
p = "components/demo/gamma/component.yaml"; s = open(p).read()
s = s.replace("    - demo/beta@0.1.0\n", "    - demo/alpha@0.2.0\n    - demo/beta@0.1.0\n", 1); open(p, "w").write(s)
p = "components/demo/gamma/main.go"; s = open(p).read()
s = s.replace('''		writeJSON(w, map[string]any{"from": "demo/gamma", "betaEndpoint": cfg["DEMO_BETA_ENDPOINT"], "beta": b})''',
'''		var a map[string]any
		if r, err := http.Get(cfg["DEMO_ALPHA_ENDPOINT"] + "/api/hello"); err == nil {
			_ = json.NewDecoder(r.Body).Decode(&a)
			r.Body.Close()
		}
		writeJSON(w, map[string]any{"from": "demo/gamma", "betaEndpoint": cfg["DEMO_BETA_ENDPOINT"], "beta": b,
			"alphaEndpoint": cfg["DEMO_ALPHA_ENDPOINT"], "alpha": a})''', 1)
open(p, "w").write(s)
PY
# the shell compiles in alpha 0.1.0 from its tag's checkout
sed -i "s#example.test/alpha => ../../../components/demo/alpha#example.test/alpha => $S/r008/src/alpha-0.1.0#" shell/demo/shell/go.mod
(cd shell/demo/shell && rm -rf vendor && go mod vendor && go vet -mod=vendor .)
(cd components/demo/gamma && go vet ./...)
# alpha comes from the git install source
python3 - "$S/r008/remotes/" <<'PY'
import sys; p = "brickkit.yaml"; s = open(p).read()
src = f"  - name: alpha-repo\n    type: git\n    baseUrl: file://{sys.argv[1]}\n"
s = s.replace("sources:\n", "sources:\n" + src, 1) if "sources:\n" in s else s + "\nsources:\n" + src
open(p, "w").write(s)
PY
echo setup ok
