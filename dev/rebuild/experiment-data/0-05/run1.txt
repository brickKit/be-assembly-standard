// infra dials PostgreSQL and NATS at the addresses in the config table.
func infra(cfg map[string]string) map[string]string {
	out := map[string]string{}
	dial := func(name, addr string) {
		c, err := net.DialTimeout("tcp", addr, 2*time.Second)
		if err != nil {
			out[name] = addr + " FAIL " + err.Error()
			return
		}
		c.Close()
		out[name] = addr + " ok"
	}
	dial("pg", net.JoinHostPort(cfg["PG_HOST"], cfg["PG_PORT"]))
	if u, err := url.Parse(cfg["NATS_URL"]); err == nil {
		dial("nats", u.Host)
	}
	return out
}'

func main() {
	raw, ok := os.LookupEnv("BRICKKIT_SERVED_MEMBERS_CONFIG")
	if !ok {
		log.Fatal("BRICKKIT_SERVED_MEMBERS_CONFIG is not set")
	}
	var members []member
	if err := json.Unmarshal([]byte(raw), &members); err != nil {
		log.Fatalf("BRICKKIT_SERVED_MEMBERS_CONFIG: %v", err)
	}
	for _, m := range members {
		mod, ok := compiled[m.ComponentID]
		if !ok {
			log.Fatalf("this shell has no module %s", m.ComponentID)
		}
		if mod.version != m.Version {
			log.Fatalf("asked to host %s@%s, but the code compiled in is %s", m.ComponentID, m.Version, mod.version)
		}
		addr := fmt.Sprintf(":%d", m.HTTPPort)
		go func(h http.Handler) { log.Fatal(http.ListenAndServe(addr, h)) }(mod.handler(m.Config))
		log.Printf("serving %s@%s on %s", m.ComponentID, m.Version, addr)
	}
	port := os.Getenv("PORT")
	if port == "" {
		port = "8080"
	}
	http.HandleFunc("/healthz", func(w http.ResponseWriter, _ *http.Request) { w.Write([]byte("ok")) })
	log.Printf("shell on :%s", port)
	log.Fatal(http.ListenAndServe(":"+port, nil))
}

PY_COMMON='
def infra(cfg):
    out = {}
    def dial(name, host, port):
        addr = f"{host}:{port}"
        try:
            socket.create_connection((host, int(port)), timeout=2).close()
            out[name] = addr + " ok"
        except OSError as e:
            out[name] = f"{addr} FAIL {e}"
    dial("pg", cfg.get("PG_HOST", ""), cfg.get("PG_PORT", "0"))
    u = urllib.parse.urlparse(cfg.get("NATS_URL", ""))
    dial("nats", u.hostname or "", u.port or 0)
    return out


def get_json(url):
    with urllib.request.urlopen(url, timeout=5) as r:
        return json.load(r)


class Base(BaseHTTPRequestHandler):
    def log_message(self, *args):
        pass

    def send_json(self, code, value):
        body = json.dumps(value).encode()
        self.send_response(code)
        self.send_header("Content-Type", "application/json")
        self.send_header("Content-Length", str(len(body)))
        self.end_headers()
        self.wfile.write(body)'
