#!/usr/bin/env bash
# 0-05 run 3 (v1.7.1): FRB-003 — a member of the local Go shell migrates from source.
set -uo pipefail
S=$(dirname "$0")/..; S=$(cd "$S" && pwd); P=$S/r005/proj; export GOCACHE=$S/gocache
run() { echo "\$ $*"; "$@" 2>&1 | sed -E 's/"time":"[^"]*"/"time":"…"/'; echo "[exit ${PIPESTATUS[0]}]"; }
start_up() { setsid brickkit up > "$1" 2>&1 & UP=$!; }
wait_ready() { local p ok; for _ in $(seq 1 600); do ok=1; for p in "$@"; do wget -qO- -T 1 "http://localhost:$p/healthz" >/dev/null 2>&1 || { ok=0; break; }; done
  [ $ok = 1 ] && { echo "ready: $*"; return 0; }; kill -0 $UP 2>/dev/null || { echo "up exited before ready"; return 1; }; sleep 0.2; done; echo "not ready"; return 1; }
wait_end() { for _ in $(seq 1 150); do kill -0 $UP 2>/dev/null || break; sleep 0.2; done; kill -0 $UP 2>/dev/null && { echo "up still running"; kill -INT $UP; }; wait $UP; echo "up exited with $?"; }
stop_up() { kill -INT $UP 2>/dev/null; wait $UP; echo "up exited with $?"; }
mig() { # <migrationCommand JSON or empty>
  cp $S/alpha.component.yaml.orig components/demo/alpha/component.yaml
  python3 - components/demo/alpha/component.yaml "$1" <<'PY'
import sys; p, cmd = sys.argv[1], sys.argv[2]; s = open(p).read()
block = 'migration:\n  command: ["/server", "migrate"]\n\n' + (f'local:\n  migrationCommand: {cmd}\n\n' if cmd else '')
open(p, 'w').write(s.replace("healthCheck:", block + "healthCheck:", 1))
PY
}
cd "$P"
echo "# 0-05 run 3 ($(date +%F)), $(brickkit version | head -1)"
cp deploy.yaml.all-local deploy.yaml
