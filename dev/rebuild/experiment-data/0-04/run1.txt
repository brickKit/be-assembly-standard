package $n

import (

// Version is the version of this module's code; a shell compares it with the version it is told to host.

	mux := http.NewServeMux()

// Command server runs $n on its own: the config table is filled from the environment.
package main

import (
	"log"
	"net/http"
	"os"
	"strings"

)

func main() {
	cfg := map[string]string{}

// Command shell hosts the members it compiles in, as told by BRICKKIT_SERVED_MEMBERS_CONFIG.
package main

import (
	"encoding/json"
	"fmt"
	"log"
	"net/http"
	"os"

	"example.test/alpha"
	"example.test/beta"
)

type member struct {
	ComponentID string            `json:"componentId"`
	Version     string            `json:"version"`
	HTTPPort    int               `json:"httpPort"`
	Config      map[string]string `json:"config"`
}

type module struct {
	version string
	handler func(map[string]string) http.Handler
}

// compiled lists the modules this binary contains, with the version of their code.
var compiled = map[string]module{
	"demo/alpha": {alpha.Version, alpha.Handler},
	"demo/beta":  {beta.Version, beta.Handler},
}

func main() {
	raw, ok := os.LookupEnv("BRICKKIT_SERVED_MEMBERS_CONFIG")
	if !ok {
		log.Fatal("BRICKKIT_SERVED_MEMBERS_CONFIG is not set")
	}
	var members []member
	if err := json.Unmarshal([]byte(raw), &members); err != nil {
		log.Fatalf("BRICKKIT_SERVED_MEMBERS_CONFIG: %v", err)
	}
	for _, m := range members {
		mod, ok := compiled[m.ComponentID]
		if !ok {
			log.Fatalf("this shell has no module %s", m.ComponentID)
		}
		if mod.version != m.Version {
			log.Fatalf("asked to host %s@%s, but the code compiled in is %s", m.ComponentID, m.Version, mod.version)
		}
		addr := fmt.Sprintf(":%d", m.HTTPPort)
		go func(h http.Handler) { log.Fatal(http.ListenAndServe(addr, h)) }(mod.handler(m.Config))
		log.Printf("serving %s@%s on %s", m.ComponentID, m.Version, addr)
	}
	http.HandleFunc("/healthz", func(w http.ResponseWriter, _ *http.Request) { w.Write([]byte("ok")) })
	log.Fatal(http.ListenAndServe(":8080", nil))
}
EOF
