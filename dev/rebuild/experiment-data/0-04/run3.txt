#!/usr/bin/env bash
# FRR-005 repro: an image-only local component whose Dockerfile and code changed; up runs the old published image without a word.
set -uo pipefail
S=$(dirname "$0")/..; S=$(cd "$S" && pwd)
run() { echo "\$ $*"; "$@" 2>&1 | sed -E 's/"time":"[^"]*"/"time":"…"/'; echo "[exit ${PIPESTATUS[0]}]"; }
echo "# FRR-005 repro ($(date +%F)), $(brickkit version | head -1), $(docker --version | cut -d, -f1)"
bash "$S/bin/r004-setup.sh" >/dev/null 2>&1 || { echo setup failed; exit 1; }
cd "$S/r004/proj"
run brickkit add --local --yes
cat > deploy.yaml <<'YAML'
target: docker

components:
  - id: demo/shell
    members:
      - id: demo/alpha
  - id: demo/beta
YAML
echo; echo "== 1. beta writes only image: (the published 0.1.0); stand in for the registry copy with a local image built from the released code =="
sed -i '/^  build:/,/^    dockerfile:/d; s/^  type: container/  type: container\n  image: demo-beta-published:0.1.0/' components/demo/beta/component.yaml
run sed -n '/^deployment:/,/^healthCheck:/p' components/demo/beta/component.yaml
run docker build -q -t demo-beta-published:0.1.0 components/demo/beta
echo; echo "== 2. the developer changes beta's code in the working copy (version unchanged, as during development) =="
sed -i 's/"from": "demo\/beta"/"from": "demo\/beta (EDITED IN WORKING COPY)"/' components/demo/beta/beta.go
run grep -n "EDITED" components/demo/beta/beta.go
run brickkit build demo/shell@0.1.0 --force
echo; echo "== 3. up (no lint, no build for beta) =="
run brickkit up --dry-run
run brickkit up
echo; echo "== 4. which code answers =="
run docker compose -p brickkit-proj exec -T demo-beta-0-1-0 wget -qO- http://localhost:8082/api/chain
run docker compose -p brickkit-proj images demo-beta-0-1-0
run docker image inspect -f "{{.Id}} created {{.Created}}" demo-beta-published:0.1.0
echo; echo "== 5. the same project after lint: the warning was there all along =="
run brickkit lint --all
